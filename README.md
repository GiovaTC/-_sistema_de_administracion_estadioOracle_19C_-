## 1. Descripción del proyecto

Sistema web Full Stack para administrar integralmente un estadio de fútbol mediante una arquitectura de microservicios frontend (Microfrontend).

El proyecto permite gestionar instalaciones deportivas, canchas, tribunas, localidades, equipos, partidos, eventos, reservas, clientes, pagos, mantenimiento, inventario y usuarios.

## 5.7. Servicio

Archivo:

`service/EstadioService.java`

```java
package com.estadio.service;

import com.estadio.entity.Estadio;
import com.estadio.repository.EstadioRepository;

import org.springframework.stereotype.Service;

import java.util.List;

@Service
public class EstadioService {

    private final EstadioRepository repository;

    public EstadioService(EstadioRepository repository) {
        this.repository = repository;
    }

    public List<Estadio> listar() {
        return repository.findAll();
    }

    public Estadio buscar(Long id) {
        return repository.findById(id)
            .orElseThrow(() ->
                new RuntimeException("Estadio no encontrado")
            );
    }

    public Estadio guardar(Estadio estadio) {
        return repository.save(estadio);
    }

    public void eliminar(Long id) {
        if (!repository.existsById(id)) {
            throw new RuntimeException(
                "Estadio no encontrado"
            );
        }

        repository.deleteById(id);
    }
}
```

## 5.8. Controlador REST

Archivo:

`controller/EstadioController.java`

```java
package com.estadio.controller;

import com.estadio.entity.Estadio;
import com.estadio.service.EstadioService;

import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/estadios")
public class EstadioController {

    private final EstadioService service;

    public EstadioController(EstadioService service) {
        this.service = service;
    }

    @GetMapping
    public List<Estadio> listar() {
        return service.listar();
    }

    @GetMapping("/{id}")
    public Estadio buscar(@PathVariable Long id) {
        return service.buscar(id);
    }

    @PostMapping
    public Estadio guardar(@RequestBody Estadio estadio) {
        return service.guardar(estadio);
    }

    @PutMapping("/{id}")
    public Estadio actualizar(
            @PathVariable Long id,
            @RequestBody Estadio estadio) {

        Estadio actual = service.buscar(id);

        actual.setNombre(estadio.getNombre());
        actual.setCiudad(estadio.getCiudad());
        actual.setDireccion(estadio.getDireccion());
        actual.setCapacidad(estadio.getCapacidad());
        actual.setEstado(estadio.getEstado());

        return service.guardar(actual);
    }

    @DeleteMapping("/{id}")
    public void eliminar(@PathVariable Long id) {
        service.eliminar(id);
    }
}
```

Archivo:

`estadio.service.ts`

```typescript
import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';

export interface Estadio {
  id?: number;
  nombre: string;
  ciudad: string;
  direccion: string;
  capacidad: number;
  estado: string;
}

@Injectable({
  providedIn: 'root'
})
export class EstadioService {

  private api =
    'http://localhost:8080/api/estadios';

  constructor(private http: HttpClient) {}

  listar(): Observable<Estadio[]> {
    return this.http.get<Estadio[]>(this.api);
  }

  buscar(id: number): Observable<Estadio> {
    return this.http.get<Estadio>(
      `${this.api}/${id}`
    );
  }

  guardar(estadio: Estadio): Observable<Estadio> {
    return this.http.post<Estadio>(
      this.api,
      estadio
    );
  }

  actualizar(
    id: number,
    estadio: Estadio
  ): Observable<Estadio> {

    return this.http.put<Estadio>(
      `${this.api}/${id}`,
      estadio
    );
  }

  eliminar(id: number): Observable<void> {
    return this.http.delete<void>(
      `${this.api}/${id}`
    );
  }
}
```

```typescript
import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import {
  EstadioService,
  Estadio
} from '../services/estadio.service';

@Component({
  selector: 'app-estadio-dashboard',
  standalone: true,
  imports: [CommonModule],
  templateUrl: './estadio-dashboard.component.html'
})
export class EstadioDashboardComponent
  implements OnInit {

  estadios: Estadio[] = [];

  constructor(
    private estadioService: EstadioService
  ) {}

  ngOnInit(): void {
    this.cargarEstadios();
  }

  cargarEstadios(): void {

    this.estadioService.listar().subscribe({
      next: (data) => {
        this.estadios = data;
      },
      error: (error) => {
        console.error(
          'Error consultando estadios',
          error
        );
      }
    });

  }
}
```

```java
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.web.servlet.config.annotation.*;

@Configuration
public class CorsConfig implements WebMvcConfigurer {

    @Override
    public void addCorsMappings(
            CorsRegistry registry) {

        registry.addMapping("/api/**")
            .allowedOrigins(
                "http://localhost:4200",
                "http://localhost:4201",
                "http://localhost:4202",
                "http://localhost:4203",
                "http://localhost:4204"
            )
            .allowedMethods(
                "GET", "POST", "PUT", "DELETE", "OPTIONS"
            )
            .allowedHeaders("*");
    }
}
```

Para producción se deben configurar los orígenes reales y coordinar CORS con Spring Security.
